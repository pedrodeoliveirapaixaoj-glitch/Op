using UnityEngine;
using UnityEngine.UI;

public class ScoreboardUI : MonoBehaviour
{
    public Text homeScoreText;
    public Text awayScoreText;
    public Text timerText;

    void Update()
    {
        if (GameManager.Instance == null)
            return;

        homeScoreText.text = GameManager.Instance.homeScore.ToString();
        awayScoreText.text = GameManager.Instance.awayScore.ToString();
        timerText.text = GameManager.Instance.GetTimer();
    }
}
